class StrictModel(BaseModel):
    model_config = ConfigDict(extra='forbid', allow_inf_nan=False)


class Epoch(StrictModel):
    epoch: int = Field(ge=1)
    train_accuracy: float = Field(ge=0, le=1)
    validation_accuracy: float = Field(ge=0, le=1)
    train_loss: float = Field(ge=0)
    validation_loss: float = Field(ge=0)


class ModelInfo(StrictModel):
    architecture: str = Field(min_length=1, max_length=150)
    embedding_size: int | None = Field(default=None, ge=1)
    optimizer: str | None = Field(default=None, max_length=80)
    learning_rate: float | None = Field(default=None, gt=0)
    batch_size: int | None = Field(default=None, ge=1)
    training_seconds: float | None = Field(default=None, ge=0)
    checkpoint: str | None = Field(default=None, max_length=200)
    dataset_version: str | None = Field(default=None, max_length=200)


class Counts(StrictModel):
    training_images: int | None = Field(default=None, ge=0)
    validation_images: int | None = Field(default=None, ge=0)


class Trial(StrictModel):
    # Positive means this student. Scores must use higher-is-more-similar convention.
    actual_positive: bool
    predicted_positive: bool
    similarity: float = Field(ge=-1, le=1)


class Run(StrictModel):
    class_id: str = Field(min_length=1, max_length=100)
    student_id: str = Field(min_length=1, max_length=100)
    name: str = Field(min_length=1, max_length=120)
    model: ModelInfo
    counts: Counts = Field(default_factory=Counts)
    history: list[Epoch] = Field(default_factory=list, max_length=2000)
    trials: list[Trial] = Field(default_factory=list, max_length=10000)
    evaluation_split: Literal['test', 'validation'] = 'test'
    threshold: float = Field(default=.45, ge=-1, le=1)
    notes: str = Field(default='', max_length=2000)

    @model_validator(mode='after')
    def ordered_epochs(self):
        epochs = [e.epoch for e in self.history]
        if epochs != sorted(set(epochs)):
            raise ValueError('Epochs must be unique and increasing')
        return self


def ratio(numerator, denominator):
    return numerator / denominator if denominator else None


def statistics(values):
    return {'count': len(values), 'mean': sum(values)/len(values) if values else None,
            'min': min(values) if values else None, 'max': max(values) if values else None}


def evaluate_trials(trials):
    tp = sum(t['actual_positive'] and t['predicted_positive'] for t in trials)
    fn = sum(t['actual_positive'] and not t['predicted_positive'] for t in trials)
    fp = sum(not t['actual_positive'] and t['predicted_positive'] for t in trials)
    tn = sum(not t['actual_positive'] and not t['predicted_positive'] for t in trials)
    precision, recall = ratio(tp,tp+fp), ratio(tp,tp+fn)
    scores = [t['similarity'] for t in trials]
    genuine = [t['similarity'] for t in trials if t['actual_positive']]
    impostor = [t['similarity'] for t in trials if not t['actual_positive']]
    roc, auc = [], None
    if genuine and impostor:
        # Tie-aware threshold sweep includes both endpoints. No fabricated AUC floor.
        roc = [{'fpr':0., 'tpr':0., 'threshold':None}]
        for cutoff in sorted(set(scores), reverse=True):
            roc.append({'threshold':cutoff, 'tpr':sum(s>=cutoff for s in genuine)/len(genuine),
                        'fpr':sum(s>=cutoff for s in impostor)/len(impostor)})
        auc = sum((b['fpr']-a['fpr'])*(b['tpr']+a['tpr'])/2 for a,b in zip(roc,roc[1:]))
    return dict(accuracy=ratio(tp+tn,len(trials)), precision=precision, recall=recall,
                f1=ratio(2*tp,2*tp+fp+fn), far=ratio(fp,fp+tn), frr=ratio(fn,fn+tp),
                confusion={'labels':['Other / unknown','Selected student'], 'matrix':[[tn,fp],[fn,tp]],
                           'tp':tp,'fp':fp,'fn':fn,'tn':tn}, attempts=len(trials),
                similarity=statistics(scores), distance=statistics([1-s for s in scores]),
                genuine=statistics(genuine), impostor=statistics(impostor), roc=roc, auc=auc)

def build_router(get_store, get_settings):
    router = APIRouter(prefix='/api/lab', tags=['Accuracy Lab'])

    def classes():
        groups = {}
        for r in get_store().timetable_rules():
            key = (r['course_name'], r.get('batch'))
            if key not in groups:
                groups[key] = dict(id=r['id'],name=r['course_name'],batch=r.get('batch'),rule_ids=[])
            groups[key]['rule_ids'].append(r['id'])
        return [dict(id='all',name='All classes',batch=None,rule_ids=None), *groups.values()]

    def scope(class_id, sid=None):
        cls = next((c for c in classes() if c['id']==class_id),None)
        if not cls:
            raise HTTPException(404,'Class not found')
        students = [s for s in get_store().students() if not cls['batch'] or s.get('batch')==cls['batch']]
        if sid is not None and not any(s['id']==sid for s in students):
            raise HTTPException(404,'Student does not belong to the selected class')
        return cls, students

    def attendance(cls, sid):
        with get_store().connect() as db:
            query = 'SELECT COUNT(*) FROM attendance a JOIN sessions s ON s.id=a.session_id WHERE a.student_id=?'
            params = [sid]
            if cls['rule_ids'] is not None:
                query += ' AND s.rule_id IN ('+','.join('?' for _ in cls['rule_ids'])+')'
                params += cls['rule_ids']
            return db.execute(query,params).fetchone()[0]

    @router.get('/classes')
    def list_classes():
        return classes()

    @router.get('/classes/{class_id}/students')
    def list_students(class_id: str):
        return [dict(id=s['id'],name=s['name'],roll_no=s['roll_no'],batch=s.get('batch'),
                     sample_count=s['sample_count']) for s in scope(class_id)[1]]

    @router.post('/runs', status_code=201)
    def save_run(body: Run):
        scope(body.class_id,body.student_id)
        rid = str(uuid.uuid4())
        with get_store().connect() as db:
            db.execute('INSERT INTO model_runs VALUES(?,?,?,?,?,?)',
                       (rid,body.class_id,body.student_id,body.name,now(),body.model_dump_json()))
        return {'id':rid,'source':'recorded'}

    @router.get('/classes/{class_id}/students/{sid}/runs')
    def list_runs(class_id: str, sid: str):
        scope(class_id,sid)
        with get_store().connect() as db:
            return [dict(r) for r in db.execute('SELECT id,name,created_at FROM model_runs WHERE class_id=? AND student_id=? ORDER BY created_at DESC',(class_id,sid))]

    def report(class_id, sid, run_id=None, source='real'):
        cls, students = scope(class_id,sid)
        student = next(s for s in students if s['id']==sid)
        created = None
        if source=='demo':
            data, provenance, rid = demo_run(class_id,sid),'demo','demo'
        else:
            with get_store().connect() as db:
                query = 'SELECT * FROM model_runs WHERE class_id=? AND student_id=?'
                params = [class_id,sid]
                if run_id:
                    query += ' AND id=?'
                    params.append(run_id)
                row = db.execute(query+' ORDER BY created_at DESC LIMIT 1',params).fetchone()
            if row:
                data,provenance,rid,created = json.loads(row['payload']),'recorded',row['id'],row['created_at']
            elif run_id:
                raise HTTPException(404,'Run not found in the selected class/student')
            else:
                cfg=get_settings()
                trials=[]
                for p in get_store().probes():
                    if p['split']!='test' or sid not in p['scores']:
                        continue
                    ranked=sorted(p['scores'].items(),key=lambda x:x[1],reverse=True)
                    best,score=ranked[0]
                    gap=score-ranked[1][1] if len(ranked)>1 else 2
                    trials.append(dict(actual_positive=p['expected']==sid,
                        predicted_positive=best==sid and score>=cfg['threshold'] and gap>=cfg['margin'],similarity=p['scores'][sid]))
                data=Run(class_id=class_id,student_id=sid,name='Current held-out recognition probes',
                    model=ModelInfo(architecture='YuNet + SFace (frozen pretrained)',embedding_size=128),
                    trials=trials,threshold=cfg['threshold'],
                    notes='Probe evaluation spans all classes because capture records have no class label. Enrollment is not model training.').model_dump()
                provenance,rid='local_probes','local-probes'
        history=data['history']
        return dict(id=rid,name=data['name'],source=provenance,created_at=created,
                    student={k:student.get(k) for k in ['id','name','roll_no','batch']},class_id=class_id,
                    history_scope='Model run (not a separate model trained for this student)',
                    evaluation_scope='Selected student versus other/unknown identities',
                    history=history,final=history[-1] if history else None,model=data['model'],
                    counts={**data['counts'],'enrollment_images':student['sample_count'],
                            'attendance_records':attendance(cls,sid)},
                    evaluation=evaluate_trials(data['trials']),threshold=data['threshold'],
                    evaluation_split=data['evaluation_split'],notes=data['notes'])

    @router.get('/classes/{class_id}/students/{sid}/metrics')
    def metrics(class_id: str,sid: str,run_id: str | None=None,source: Literal['real','demo']='real'):
        return report(class_id,sid,run_id,source)

    @router.get('/classes/{class_id}/summary')
    def summary(class_id: str):
        rows=[]
        for s in scope(class_id)[1]:
            r=report(class_id,s['id'])
            rows.append(dict(student_id=s['id'],name=s['name'],roll_no=s['roll_no'],source=r['source'],
                             run_name=r['name'],attempts=r['evaluation']['attempts'],
                             precision=r['evaluation']['precision'],recall=r['evaluation']['recall'],
                             f1=r['evaluation']['f1'],far=r['evaluation']['far'],frr=r['evaluation']['frr']))
        return rows

    return router

def _local_hostname(host_header: str) -> str:
    host = (host_header or '').strip()
    if host.startswith('['):
        end = host.find(']')
        return host[1:end] if end != -1 else host
    return host.split(':')[0]


def _allow_local_host(host_header: str) -> bool:
    hostname = _local_hostname(host_header).lower()
    if hostname in {'localhost', '127.0.0.1', '::1', 'testserver'} or hostname.endswith('.local'):
        return True
    try:
        return ipaddress.ip_address(hostname).is_private
    except ValueError:
        return False


@app.middleware('http')
async def require_local_host(request: Request, call_next):
    if not _allow_local_host(request.headers.get('host', '')):
        return JSONResponse({'detail': 'Host not allowed'}, status_code=400)
    return await call_next(request)

@app.middleware('http')
async def local_only(request: Request, call_next):
    origin = request.headers.get('origin')
    if origin and origin != str(request.base_url).rstrip('/'):
        return JSONResponse({'detail': 'Cross-origin access is disabled'}, status_code=403)
    if int(request.headers.get('content-length', '0')) > 10_000_000:
        return JSONResponse({'detail': 'Request too large'}, status_code=413)
    response = await call_next(request)
    response.headers['X-Content-Type-Options'] = 'nosniff'
    response.headers['Referrer-Policy'] = 'no-referrer'
    response.headers['Cache-Control'] = 'no-store'
    response.headers['Content-Security-Policy'] = (
        "default-src 'self'; img-src 'self' data:; media-src 'self' blob:; "
        "style-src 'self' 'unsafe-inline'; script-src 'self'; "
        "connect-src 'self'; frame-ancestors 'none'")
    return response

def settings():
    return store.setting('calibration',

def get_current_classes():
    """Determine all classes currently in session based on the timetable."""
    today = datetime.now()
    weekday = today.weekday()  # 0=Mon..6=Sun
    if weekday > 5:
        return []
    current_time = today.strftime('%H:%M')
    date_str = today.strftime('%Y-%m-%d')
    schedule = store.today_schedule(weekday, date_str)
    return [rule for rule in schedule
            if rule['start_time'] <= current_time < rule['end_time']]


def get_current_session_info():
    """Determine what class should be happening right now based on timetable."""
    classes = get_current_classes()
    return classes[0] if classes else None


def get_next_class():
    """Find the next upcoming class today."""
    today = datetime.now()
    weekday = today.weekday()
    if weekday > 5:
        return None
    current_time = today.strftime('%H:%M')
    date_str = today.strftime('%Y-%m-%d')
    schedule = store.today_schedule(weekday, date_str)
    for rule in schedule:
        if rule['start_time'] > current_time:
            return rule
    return None


def compute_dl_metrics():
    cfg = settings()
    threshold = float(cfg['threshold'])
    margin = float(cfg['margin'])
    students = [s for s in store.students(True) if s.get('centroid')]
    stats = {}
    n = len(students)
    if n > 0:
        centroids = np.array([s['centroid'] for s in students], dtype=np.float32)
        sim_matrix = np.clip(centroids @ centroids.T, -1.0, 1.0)
        impostor_scores = []
        for i, s in enumerate(students):
            other_scores = [float(sim_matrix[i, j]) for j in range(n) if j != i]
            max_imp = max(other_scores) if other_scores else 0.0
            avg_imp = float(np.mean(other_scores)) if other_scores else 0.0
            impostor_scores.extend(other_scores)
            stats[s['id']] = {
                'max_impostor': round(max_imp, 4),
                'avg_impostor': round(avg_imp, 4),
                'margin_to_threshold': round(threshold - max_imp, 4),
                'validation_pass': bool(max_imp < threshold),
                'self_similarity': 1.0,
                'sample_count': s.get('sample_count', 0),
            }
        max_imp_overall = max(impostor_scores) if impostor_scores else 0.0
        avg_imp_overall = float(np.mean(impostor_scores)) if impostor_scores else 0.0
        far = float(np.mean([s >= threshold for s in impostor_scores])) if impostor_scores else 0.0
        acc = float(np.mean([s < threshold for s in impostor_scores])) if impostor_scores else 1.0
    else:
        max_imp_overall = 0.0
        avg_imp_overall = 0.0
        far = 0.0
        acc = 1.0

    eval_data = store.probes()
    val_rows = [r for r in eval_data if r['split'] == 'validation']
    test_rows = [r for r in eval_data if r['split'] == 'test']

    return {
        'model_detector': 'YuNet 2023mar (ONNX)',
        'model_recognizer': 'SFace 2021dec (128-d L2)',
        'embedding_dims': 128,
        'distance_metric': 'Cosine Similarity',
        'threshold': threshold,
        'margin': margin,
        'calibrated': bool(cfg['calibrated']),
        'total_enrolled': n,
        'max_impostor_similarity': round(max_imp_overall, 4) if n > 1 else None,
        'avg_impostor_similarity': round(avg_imp_overall, 4),
        'gallery_separation_rate': round(acc, 4) if n > 1 else None,
        'pairwise_far': round(far, 4),
        'validation_attempts': len(val_rows),
        'test_attempts': len(test_rows),
        'student_metrics': stats,
    }

@app.get('/api/state')
def state():
    models_ready = all(
        (ROOT / 'models' / f).exists()
        for f in ('face_detection_yunet_2023mar.onnx',
                  'face_recognition_sface_2021dec.onnx'))
    active = store.active_session()
    current_classes = get_current_classes()
    if active and active.get('rule_id'):
        matched_rule = next((r for r in store.timetable_rules()
                             if r['id'] == active['rule_id']), None)
        current_class = matched_rule or (current_classes[0] if current_classes else None)
    else:
        current_class = current_classes[0] if current_classes else None
    next_class = get_next_class() if not current_classes else None
    return dict(
        students=store.students(),
        sessions=store.sessions(),
        records=store.all_records(),
        settings=settings(),
        models_ready=models_ready,
        timetable=store.timetable_rules(),
        exceptions=store.timetable_exceptions(),
        active_session=active,
        current_class=current_class,
        current_classes=current_classes,
        next_class=next_class,
        dl_metrics=compute_dl_metrics(),
        today=datetime.now().strftime('%Y-%m-%d'),
        weekday=datetime.now().weekday(),
        day_name=DAY_NAMES[datetime.now().weekday()]

@app.post('/api/students/{sid}/enrollment/sample')
def enrollment_sample(sid: str, body: ConsentFrame):
    """Capture a single enrollment sample for a student."""
    with lock:
        # Verify student exists
        store.get_student(sid)
        result = engine.extract(body.image)
        embedding = result.pop('embedding', None)
        if embedding is not None:
            stamp = time.monotonic()
            for key in list(pending):
                if stamp - pending[key][0] > 600:
                    del pending[key]
            if len(pending) >= 200:
                raise ValueError(
                    'Too many pending samples. Wait ten minutes or restart.')
            token = str(uuid.uuid4())
            pending[token] = (stamp, embedding)
            result['token'] = token
        return result


@app.post('/api/students/{sid}/enrollment/save')
def enrollment_save(sid: str, body: EnrollmentSave):
    """Save enrollment embeddings for a student."""
    with lock:
        store.get_student(sid)
        if len(set(body.tokens)) != len(body.tokens):
            raise ValueError('Capture distinct samples')
        if any(t not in pending or time.monotonic() - pending[t][0] > 600
               for t in body.tokens):
            raise ValueError('Enrollment captures expired. Capture again.')
        vectors = [pending[t][1] for t in body.tokens]
        centroid = normalize(np.mean(vectors, axis=0))
        if min(float(np.dot(v, centroid)) for v in vectors) < .45:
            raise ValueError(
                'Samples are inconsistent. Capture the same person in even '
                'lighting.')
        # Check for duplicate enrollment
        gallery = {s['id']: s['centroid']
                   for s in store.students(True) if s.get('centroid')}
        if match(centroid, gallery, .65, 0)['identity']:
            raise ValueError(
                'These samples closely match an enrolled student. '
                'Check for duplicate enrollment.')
        store.update_student_enrollment(sid, vectors, centroid.tolist())
        for t in body.tokens:
            del pending[t]
        stability.clear()
    return {'ok': True}


@app.post('/api/students/{sid}/upload')
async def upload_enrollment(sid: str, file: UploadFile = File(...),
                            consent: str = Form('true')):
    """Upload photo(s) or video for enrollment from device."""
    if consent.lower() != 'true':
        raise ValueError('Consent is required')
    with lock:
        store.get_student(sid)
    content = await file.read()
    if len(content) > 10_000_000:
        raise ValueError('File too large. Maximum 10 MB.')

    import base64
    content_type = file.content_type or ''

    if content_type.startswith('video/'):
        # Extract frames from video
        vectors = _extract_from_video(content)
    else:
        # Treat as image
        encoded = base64.b64encode(content).decode()
        result = engine.extract(encoded)
        embedding = result.pop('embedding', None)
        if embedding is None:
            raise ValueError(result.get('reason', 'No face detected in image'))
        vectors = [embedding]

    if len(vectors) < 1:
        raise ValueError('No usable face frames found in the uploaded file.')

    centroid = normalize(np.mean(vectors, axis=0))
    if len(vectors) > 1:
        min_sim = min(float(np.dot(v, centroid)) for v in vectors)
        if min_sim < .4:
            raise ValueError(
                'Frames are too inconsistent. Use a clearer video of one person.')

    with lock:
        gallery = {s['id']: s['centroid']
                   for s in store.students(True) if s.get('centroid')}
        existing_match = match(centroid, gallery, .65, 0)['identity']
        if existing_match and existing_match != sid:
            raise ValueError(
                'This face closely matches another enrolled student.')
        store.update_student_enrollment(sid, vectors, centroid.tolist())
        stability.clear()

    return {'ok': True, 'samples': len(vectors)}


def _extract_from_video(content: bytes):
    """Extract face embeddings from video bytes by sampling frames."""
    import tempfile
    import cv2 as cv

    with tempfile.NamedTemporaryFile(suffix='.mp4', delete=False) as tmp:
        tmp.write(content)
        tmp_path = tmp.name

    try:
        cap = cv.VideoCapture(tmp_path)
        if not cap.isOpened():
            raise ValueError('Cannot open video file')

        fps = cap.get(cv.CAP_PROP_FPS) or 30
        total = int(cap.get(cv.CAP_PROP_FRAME_COUNT))
        duration = total / fps if fps > 0 else 0
        if duration > 60:
            raise ValueError('Video must be 60 seconds or shorter.')

        # Sample one frame every second
        sample_interval = max(1, int(fps))
        vectors = []
        frame_idx = 0

        while True:
            ret, frame = cap.read()
            if not ret:
                break
            if frame_idx % sample_interval == 0:
                import base64
                _, buf = cv.imencode('.jpg', frame)
                encoded = base64.b64encode(buf.tobytes()).decode()
                try:
                    result = engine.extract(encoded)
                    emb = result.get('embedding')
                    if emb is not None:
                        vectors.append(emb)
                except Exception:
                    pass
                if len(vectors) >= 15:
                    break
            frame_idx += 1
        cap.release()
    finally:
        Path(tmp_path).unlink(missing_ok=True)

    return vectors

@app.post('/api/sessions/{sid}/end')
def end_session(sid: str):
    with lock:
        # Mark absent remaining students for this session
        session = store.active_session()
        if session and session['id'] == sid and session.get('rule_id'):
            # Find eligible students based on the timetable rule
            rule = None
            for r in store.timetable_rules():
                if r['id'] == session['rule_id']:
                    rule = r
                    break
            if rule:
                all_students = store.students()
                if rule.get('batch') and rule['batch'] in ('D1', 'D2', 'D3'):
                    eligible = [s['id'] for s in all_students
                                if s['batch'] == rule['batch']]
                else:
                    eligible = [s['id'] for s in all_students]
                store.mark_absent_remaining(sid, eligible)
        store.end_session(sid)
        stability.clear()
    return {'ok': True}

@app.post('/api/recognize')
def recognize(body: Frame):
    with lock:
        result = engine.extract(body.image)
        embedding = result.pop('embedding', None)
        students = store.students(True)
        gallery = {s['id']: s['centroid']
                   for s in students if s.get('centroid')}
        config = settings()
        decision = (match(embedding, gallery, config['threshold'],
                          config['margin'])
                    if embedding else
                    dict(identity=None, score=None, gap=None))
        decision.pop('scores', None)
        key = (body.stream_id, body.session_id)
        if len(stability) > 100:
            stability.clear()
        stab_obj = stability.setdefault(key, Stability())
        stable = stab_obj.observe(decision['identity'], time.monotonic())
        stable_count = getattr(stab_obj, 'count', 0)
        marked = False
        if stable and body.session_id:
            marked = store.mark(
                decision['identity'], body.session_id, decision['score'])
        name = next((s['name'] for s in students
                     if s['id'] == decision['identity']), None)
        roll = next((s['roll_no'] for s in students
                     if s['id'] == decision['identity']), None)
        return {**result, **decision,
                'stable': stable, 'marked': marked,
                'name': name, 'roll_no': roll,
                'stable_count': min(stable_count, 3),
                'threshold': config['threshold'],
                'margin': config['margin'],
                'embedding_dim': 128 if embedding is not None else None}

@app.post('/api/probes')
def add_probe(body: Probe):
    with lock:
        config = settings()
        if body.split == 'test' and not config['calibrated']:
            raise ValueError(
                'Calibrate using validation attempts before collecting '
                'the held-out test set')
        if body.split == 'validation' and config['calibrated']:
            raise ValueError(
                'Validation is frozen after calibration. '
                'Reset the experiment to collect more.')
        students = store.students(True)
        if not students:
            raise ValueError('Enroll students before evaluating')
        if body.expected and body.expected not in {s['id'] for s in students}:
            raise ValueError('Student not found')
        if not body.subject.strip() or not body.batch.strip():
            raise ValueError('Subject code and capture batch are required')
        result = engine.extract(body.image)
        embedding = result.pop('embedding', None)
        if embedding is None:
            raise ValueError(result['reason'])
        gallery = {s['id']: s['centroid']
                   for s in students if s.get('centroid')}
        scores = match(embedding, gallery)['scores']
        store.add_probe(body.split, body.subject.strip(), body.batch.strip(),
                        body.expected, scores, result['latency_ms'])
        return result


def evaluate(rows, threshold, margin):
    known = [r for r in rows if r['expected']]
    unknown = [r for r in rows if not r['expected']]
    genuine, impostor = [], []
    correct = wrong = rejected = unknown_correct = 0
    for r in rows:
        ranked = sorted(r['scores'].items(), key=lambda x: x[1], reverse=True)
        best, score = ranked[0]
        gap = score - ranked[1][1] if len(ranked) > 1 else 2
        prediction = best if score >= threshold and gap >= margin else None
        if r['expected']:
            genuine.append(r['scores'][r['expected']])
            correct += prediction == r['expected']
            wrong += prediction is not None and prediction != r['expected']
            rejected += prediction is None
        else:
            unknown_correct += prediction is None
        impostor.extend(v for k, v in r['scores'].items()
                        if k != r['expected'])
    latencies = [r['latency'] for r in rows]
    return dict(
        attempts=len(rows), known_attempts=len(known),
        unknown_attempts=len(unknown),
        recognition_accuracy=correct / len(known) if known else None,
        unknown_rejection_accuracy=(unknown_correct / len(unknown)
                                    if unknown else None),
        wrong_identity_count=wrong, known_rejected=rejected,
        verification=verification_metrics(genuine, impostor, threshold),
        latency_median_ms=float(np.median(latencies)) if latencies else None,
        latency_p95_ms=float(np.percentile(latencies, 95))
        if latencies else None)


@app.get('/api/evaluation')
def evaluation():
    config = settings()
    rows = store.probes()
    return dict(
        settings=config,
        validation=evaluate(
            [r for r in rows if r['split'] == 'validation'],
            config['threshold'], config['margin']),
        test=evaluate(
            [r for r in rows if r['split'] == 'test'],
            config['threshold'], config['margin']))


@app.post('/api/calibrate')
def calibration():
    with lock:
        if settings()['calibrated']:
            raise ValueError(
                'Calibration is locked. Reset the experiment to recalibrate.')
        rows = [r for r in store.probes() if r['split'] == 'validation']
        known = [r for r in rows if r['expected']]
        unknown = [r for r in rows if not r['expected']]
        if len(known) < 5 or len(unknown) < 5:
            raise ValueError(
                'Collect at least 5 known and 5 unknown validation attempts; '
                'more are recommended.')
        genuine = [r['scores'][r['expected']] for r in known]
        impostor = [max(v for k, v in r['scores'].items()
                        if k != r['expected'])
                    for r in rows
                    if any(k != r['expected'] for k in r['scores'])]
        threshold = calibrate(genuine, impostor, 0.01)
        config = dict(threshold=threshold, margin=.08, calibrated=True,
                      created_at=now(), validation_attempts=len(rows),
                      target_far=.01)
        store.set_setting('calibration', config)
        stability.clear()
        return config

import json

def now():
    return datetime.now(timezone.utc).isoformat()


class Store:
    def __init__(self, path):
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.connect() as db:
            db.executescript('''
            PRAGMA journal_mode=WAL;

            CREATE TABLE IF NOT EXISTS students(
                id TEXT PRIMARY KEY,
                roll_no TEXT UNIQUE NOT NULL,
                name TEXT NOT NULL,
                email TEXT,
                section TEXT NOT NULL DEFAULT 'D',
                batch TEXT,
                photo_path TEXT,
                consent_at TEXT,
                samples TEXT,
                centroid TEXT
            );

            CREATE TABLE IF NOT EXISTS timetable_rules(
                id TEXT PRIMARY KEY,
                weekday INTEGER NOT NULL,
                start_time TEXT NOT NULL,
                end_time TEXT NOT NULL,
                course_name TEXT NOT NULL,
                course_code TEXT,
                teacher TEXT,
                room TEXT,
                batch TEXT,
                is_lab INTEGER NOT NULL DEFAULT 0
            );

            CREATE TABLE IF NOT EXISTS timetable_exceptions(
                id TEXT PRIMARY KEY,
                rule_id TEXT REFERENCES timetable_rules(id) ON DELETE CASCADE,
                date TEXT NOT NULL,
                action TEXT NOT NULL DEFAULT 'cancel',
                replacement TEXT,
                reason TEXT,
                created_at TEXT NOT NULL
            );

            CREATE TABLE IF NOT EXISTS sessions(
                id TEXT PRIMARY KEY,
                rule_id TEXT REFERENCES timetable_rules(id),
                name TEXT NOT NULL,
                date TEXT,
                started_at TEXT NOT NULL,
                ended_at TEXT
            );
            CREATE UNIQUE INDEX IF NOT EXISTS one_active ON sessions((1)) WHERE ended_at IS NULL;

            CREATE TABLE IF NOT EXISTS attendance(
                student_id TEXT REFERENCES students(id) ON DELETE CASCADE,
                session_id TEXT REFERENCES sessions(id),
                status TEXT NOT NULL DEFAULT 'present',
                similarity REAL,
                marked_at TEXT NOT NULL,
                method TEXT NOT NULL DEFAULT 'face',
                PRIMARY KEY(student_id, session_id)
            );

            CREATE TABLE IF NOT EXISTS settings(key TEXT PRIMARY KEY, value TEXT NOT NULL);

            CREATE TABLE IF NOT EXISTS model_runs(
                id TEXT PRIMARY KEY,
                class_id TEXT NOT NULL,
                student_id TEXT NOT NULL REFERENCES students(id) ON DELETE CASCADE,
                name TEXT NOT NULL,
                created_at TEXT NOT NULL,
                payload TEXT NOT NULL
            );
            CREATE INDEX IF NOT EXISTS model_run_scope ON model_runs(class_id, student_id, created_at);

            CREATE TABLE IF NOT EXISTS probes(
                id TEXT PRIMARY KEY,
                split TEXT,
                subject TEXT,
                batch TEXT,
                expected TEXT,
                scores TEXT,
                latency REAL,
                created_at TEXT
            );
            ''')

    def connect(self):
        db = sqlite3.connect(self.path, timeout=10)
        db.row_factory = sqlite3.Row
        db.execute('PRAGMA foreign_keys=ON')
        return db

    # â”€â”€ Students â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

    def students(self, vectors=False):
        with self.connect() as db:
            rows = [dict(r) for r in db.execute('SELECT * FROM students ORDER BY roll_no')]
        for r in rows:
            if r.get('samples'):
                r['sample_count'] = len(json.loads(r['samples']))
            else:
                r['sample_count'] = 0
            if vectors and r.get('centroid'):
                r['centroid'] = json.loads(r['centroid'])
            else:
                r.pop('centroid', None)
            r.pop('samples', None)

        def _roll_key(s):
            m = re.search(r'\d+', s.get('roll_no') or '')
            return int(m.group(0)) if m else 9999

        rows.sort(key=_roll_key)
        return rows

    def get_student(self, sid):
        with self.connect() as db:
            row = db.execute('SELECT * FROM students WHERE id=?', (sid,)).fetchone()
        if not row:
            raise ValueError('Student not found')
        r = dict(row)
        if r.get('samples'):
            r['sample_count'] = len(json.loads(r['samples']))
        else:
            r['sample_count'] = 0
        r.pop('samples', None)
        r.pop('centroid', None)
        return r

    def add_student(self, roll_no, name, email=None, section='D', batch=None,
                    photo_path=None, samples=None, centroid=None, consent_at=None):
        pid = str(uuid.uuid4())
        with self.connect() as db:
            db.execute(
                'INSERT INTO students VALUES(?,?,?,?,?,?,?,?,?,?)',
                (pid, roll_no, name, email, section, batch, photo_path,
                 consent_at or (now() if samples else None),
                 json.dumps(samples) if samples else None,
                 json.dumps(centroid) if centroid else None)
            )
        return pid

    def update_student_enrollment(self, sid, samples, centroid, photo_path=None):
        """Update face embeddings for an existing student."""
        with self.connect() as db:
            updates = {'samples': json.dumps(samples), 'centroid': json.dumps(centroid),
                       'consent_at': now()}
            if photo_path:
                updates['photo_path'] = photo_path
            sets = ', '.join(f'{k}=?' for k in updates)
            db.execute(f'UPDATE students SET {sets} WHERE id=?',
                       (*updates.values(), sid))
            self.invalidate(db)

    def invalidate(self, db):
        db.execute('DELETE FROM probes')
        db.execute("DELETE FROM settings WHERE key='calibration'")

    def delete_student(self, sid):
        with self.connect() as db:
            if not db.execute('DELETE FROM students WHERE id=?', (sid,)).rowcount:

import base64
import io
import threading
import time
from pathlib import Path
import cv2
import numpy as np

from .core import normalize

ROOT = Path(__file__).resolve().parents[1]


class FaceEngine:
    def __init__(self):
        self.lock = threading.Lock()
        self.detector = None
        self.recognizer = None

    def load(self):
        if self.detector is None:
            detector = ROOT/'models/face_detection_yunet_2023mar.onnx'
            recognizer = ROOT/'models/face_recognition_sface_2021dec.onnx'
            if not detector.exists() or not recognizer.exists():
                raise ValueError('Models missing. Run python scripts/download_models.py')
            cv2.setNumThreads(2)
            self.detector = cv2.FaceDetectorYN.create(str(detector), '', (640,480), .9, .3, 5000)
            self.recognizer = cv2.FaceRecognizerSF.create(str(recognizer), '')

    def extract(self, encoded):
        start = time.perf_counter()
        try:
            raw = base64.b64decode(encoded.split(',')[-1], validate=True)
            with Image.open(io.BytesIO(raw)) as im:
                if im.width * im.height > 1920*1080 or min(im.size) < 32:
                    raise ValueError('Image must be between 32 pixels and 1920 x 1080')
                frame = cv2.cvtColor(np.asarray(im.convert('RGB')), cv2.COLOR_RGB2BGR)
        except Exception as e:
            raise ValueError('Invalid image. Send a JPEG/PNG frame up to 1920 x 1080.') from e
        h,w = frame.shape[:2]
        result = dict(face_count=0, box=None, quality=False, reason='No face detected', width=w, height=h)
        with self.lock:
            self.load()
            self.detector.setInputSize((w,h))
            _, faces = self.detector.detect(frame)
            count = 0 if faces is None else len(faces)
            result['face_count'] = count
            if count != 1:
                result['reason'] = 'Only one person should be in frame' if count > 1 else 'No face detected'
            else:
                face = faces[0]
                x,y,fw,fh = [int(t) for t in face[:4]]
                result['box'] = [x,y,fw,fh]
                if min(fw,fh) < 80:
                    result['reason'] = 'Move closer to the camera'
                elif x < 0 or y < 0 or x+fw > w or y+fh > h:
                    result['reason'] = 'Keep your entire face in frame'
                else:
                    crop = frame[y:y+fh,x:x+fw]
                    gray = cv2.cvtColor(cv2.resize(crop,(112,112)), cv2.COLOR_BGR2GRAY)
                    blur = float(cv2.Laplacian(gray,cv2.CV_64F).var())
                    brightness = float(gray.mean())
                    result.update(blur=blur,brightness=brightness)
                    if blur < 35:
                        result['reason'] = 'Hold still; the face is blurry'
                    elif not 40 < brightness < 220:
                        result['reason'] = 'Use even, brighter lighting without backlight'
                    else:
                        aligned = self.recognizer.alignCrop(frame,face)
                        result.update(quality=True,reason='Face quality passed',embedding=normalize(self.recognizer.feature(aligned)).tolist())
        result['latency_ms'] = round((time.perf_counter()-start)*1000,1)
        return result

export async function api(path, options = {}) {
  const response = await fetch(`/api${path}`, {
    ...options,
    headers: { 'Content-Type': 'application/json', ...options.headers },
    body: options.body === undefined ? undefined : JSON.stringify(options.body),
  });
  let data = null;
  try {
    data = await response.json();
  } catch {
    if (!response.ok) {
      throw new Error(`Service unavailable (${response.status}). Please verify the backend is running on port 8000.`);
    }
    throw new Error('Invalid response received from the server.');
  }
  if (!response.ok) throw new Error(typeof data?.detail === 'string' ? data.detail : 'Please check the fields and try again.');
  return data;
}

function Schedule({state, refresh, action, setToast, setError}) {
  const [activeTab, setActiveTab] = useState('weekly');
  const [editRule, setEditRule] = useState(null);
  const [editForm, setEditForm] = useState({});
  const [editBusy, setEditBusy] = useState(false);
  const [cancelModal, setCancelModal] = useState(null);
  const [cancelDate, setCancelDate] = useState(state.today || '');
  const [cancelReason, setCancelReason] = useState('');
  const timetable = state.timetable || [];
  const exceptions = state.exceptions || [];

  const grouped = {};
  for (const rule of timetable) {
    const day = rule.weekday;
    if (!grouped[day]) grouped[day] = [];
    grouped[day].push(rule);
  }
  for (const day in grouped) {
    grouped[day].sort((a,b) => a.start_time.localeCompare(b.start_time));
  }

  // Collect all unique time slots across all days for the grid rows
  const allSlots = [];
  const slotSet = new Set();
  for (const rule of timetable) {
    const key = `${rule.start_time}-${rule.end_time}`;
    if (!slotSet.has(key)) {
      slotSet.add(key);
      allSlots.push({start: rule.start_time, end: rule.end_time});
    }
  }
  allSlots.sort((a,b) => a.start.localeCompare(b.start));

  function openEdit(rule) {
    setEditRule(rule);
    setEditForm({
      course_name: rule.course_name,
      course_code: rule.course_code || '',
      teacher: rule.teacher || '',
      room: rule.room || '',
      start_time: rule.start_time,
      end_time: rule.end_time,
      batch: rule.batch || '',
      is_lab: !!rule.is_lab,
    });
  }

  async function saveEdit(e) {
    e.preventDefault();
    setEditBusy(true);
    try {
      await api(`/timetable/${editRule.id}`, {method:'PUT', body: editForm});
      await refresh();
      setEditRule(null);
      setToast('Timetable updated.');
    } catch(err) { setError(err.message); }
    finally { setEditBusy(false); }
  }

  async function cancelClass(e) {
    e.preventDefault();
    try {
      await api('/timetable/exceptions', {method:'POST', body:{
        rule_id: cancelModal.id,
        date: cancelDate,
        action: 'cancel',
        reason: cancelReason || null
      }});
      await refresh();
      setCancelModal(null);
      setCancelReason('');
      setToast('Class cancelled for that date.');
    } catch(err) { setError(err.message); }
  }

  function getClassesForSlot(day, slot) {
    return (grouped[day] || []).filter(r => r.start_time === slot.start && r.end_time === slot.end);
  }

  return <>
    <div className="tab-bar">
      <button className={activeTab === 'weekly' ? 'tab active' : 'tab'} onClick={() => setActiveTab('weekly')}>
        <BookOpen size={15}/>Weekly timetable
      </button>
      <button className={activeTab === 'exceptions' ? 'tab active' : 'tab'} onClick={() => setActiveTab('exceptions')}>
        <AlertTriangle size={15}/>Exceptions ({exceptions.length})
      </button>
    </div>

    {activeTab === 'weekly' && <>
      <div className="tt-grid-wrap">

          <thead>

              <th className="tt-time-col">Time</th>

                  {state.weekday === d && <span className="today-badge">Today</span>}
                </th>
              )}
            </tr>
          </thead>
          <tbody>
            {allSlots.map(slot => {
              const slotKey = `${slot.start}-${slot.end}`;
              return <tr key={slotKey}>
                <td className="tt-time-cell">
                  <span className="tt-time-start">{slot.start}</span>
                  <span className="tt-time-sep">â€“</span>
                  <span className="tt-time-end">{slot.end}</span>
                </td>
                {[0,1,2,3,4,5].map(day => {
                  const classes = getClassesForSlot(day, slot);
                  return <td key={day} className={`tt-cell ${state.weekday === day ? 'tt-today-col' : ''}`}>
                    {classes.length ? classes.map(rule =>
                      <div key={rule.id} className={`tt-card ${rule.is_lab ? 'tt-lab' : 'tt-lecture'}`}>
                        <div className="tt-card-name">{rule.course_name}</div>
                        <div className="tt-card-info">
                          {rule.teacher && <span>{rule.teacher}</span>}
                          {rule.room && <span>ðŸ“ {rule.room}</span>}
                        </div>
                        <div className="tt-card-footer">
                          <span className="batch-tag">{rule.batch || 'All D'}</span>
                          <div className="tt-card-actions">
                            <button className="tt-action-btn" title="Edit class" onClick={() => openEdit(rule)}><Pencil size={12}/></button>
                            <button className="tt-action-btn tt-cancel-btn" title="Cancel on a date" onClick={() => { setCancelModal(rule); setCancelDate(state.today); }}><X size={12}/></button>
                          </div>
                        </div>
                      </div>
                    ) : <div className="tt-empty">â€”</div>}
                  </td>;
                })}
              </tr>;
            })}
          </tbody>
        </table>
      </div>
      <div className="tt-legend">
        <span><i className="tt-dot tt-lecture-dot"/>Lecture</span>
        <span><i className="tt-dot tt-lab-dot"/>Lab</span>
      </div>
    </>}

    {activeTab === 'exceptions' && <section className="panel">
      <div className="panel-heading"><h2>Schedule exceptions</h2></div>

            <td><b>{e.date}</b></td>
            <td><StatusBadge status={e.action === 'cancel' ? 'absent' : 'present'}/></td>
            <td>{e.reason || 'â€”'}</td>
            <td><button className="icon-button delete" aria-label="Remove exception" onClick={() => action(() => api(`/timetable/exceptions/${e.id}`, {method:'DELETE'}), 'Exception removed.')}><Trash2 size={17}/></button></td>
          </tr>
        )}</tbody>

          <label>Course code<input value={editForm.course_code} onChange={e => setEditForm({...editForm, course_code: e.target.value})}/></label>
          <div className="form-row">

            <button type="button" className="outline" onClick={() => setEditRule(null)}>Cancel</button>
            <button disabled={editBusy}>{editBusy ? 'Savingâ€¦' : 'Save changes'}</button>
          </div>
        </form>
      </section>
    </div>}

    {/* Cancel modal */}
    {cancelModal && <div className="modal-backdrop" onClick={() => setCancelModal(null)}>
      <section className="modal" onClick={e => e.stopPropagation()}>
        <h2>Cancel class</h2>
        <p>Cancel <b>{cancelModal.course_name}</b> ({DAY_NAMES[cancelModal.weekday]} {cancelModal.start_time}â€“{cancelModal.end_time}) on a specific date.</p>

          <label>Date<input type="date" value={cancelDate} onChange={e => setCancelDate(e.target.value)} required/></label>

function Enrollment({state, onDone, back, setError}) {
  const [studentId, setStudentId] = useState('');
  const [consent, setConsent] = useState(false);
  const [tokens, setTokens] = useState([]);
  const [result, setResult] = useState(null);
  const [saving, setSaving] = useState(false);
  const [localError, setLocalError] = useState('');
  const [uploadMode, setUploadMode] = useState('camera');
  const [uploading, setUploading] = useState(false);
  const fileRef = useRef(null);

  const unenrolledStudents = (state?.students || []).filter(s => s.sample_count === 0);
  const selectedStudent = (state?.students || []).find(s => s.id === studentId);

  const prompts = [
    'Look straight ahead in even lighting',
    'Turn slightly to your left',
    'Turn slightly to your right',
    'Raise your chin slightly',
    'Lower your chin slightly',
    'Try a modest change in lighting',
  ];

  async function capture(image) {
    if (!studentId) throw new Error('Select a student first');
    const r = await api(`/students/${studentId}/enrollment/sample`, {method:'POST', body:{image, consent}});
    setResult(r);
    if (!r.token) throw new Error(r.reason);
    setTokens(v => [...v, r.token]);
  }

  async function handleFileUpload(e) {
    const files = e.target.files;
    if (!files?.length || !studentId) return;
    setUploading(true);
    setLocalError('');
    try {
      for (const file of files) {
        const fd = new FormData();
        fd.append('file', file);
        fd.append('consent', 'true');
        const r = await uploadFile(`/students/${studentId}/upload`, fd);
        if (r.ok) {
          setTokens(prev => [...prev, ...(new Array(r.samples || 1).fill('uploaded'))]);
        }
      }
      await onDone();
    } catch (e) {
      setLocalError(e.message);
    } finally {
      setUploading(false);
    }
  }

  async function save(e) {
    e.preventDefault();
    setSaving(true);
    setLocalError('');
    try {
      await api(`/students/${studentId}/enrollment/save`, {method:'POST', body:{consent, tokens: tokens.filter(t => t !== 'uploaded')}});
      await onDone();
    } catch (e) {
      setLocalError(e.message);
    } finally {
      setSaving(false);
    }
  }

  const realTokens = tokens.filter(t => t !== 'uploaded');
  const hasUploaded = tokens.some(t => t === 'uploaded');

  return <>
    <button className="back-link" onClick={back}><ArrowLeft size={16}/>Back to students</button>
    <div className="workspace enrollment">
      <div className="enroll-left">
        {/* Mode toggle */}
        <div className="upload-toggle">
          <button className={uploadMode === 'camera' ? 'active' : ''} onClick={() => setUploadMode('camera')}><CameraIcon size={16}/>Live camera</button>
          <button className={uploadMode === 'upload' ? 'active' : ''} onClick={() => setUploadMode('upload')}><Upload size={16}/>Upload from device</button>
        </div>

        {uploadMode === 'camera' ?
          <Camera onFrame={capture} enabled={consent && !!studentId && tokens.length < 15} result={result} actionLabel={`Capture sample ${tokens.length + 1}`}/>
        :
          <section className="panel upload-panel">
            <div className="upload-zone" onClick={() => consent && studentId && fileRef.current?.click()}>
              <input ref={fileRef} type="file" accept="image/*,video/*" multiple onChange={handleFileUpload} className="hidden"/>
              {uploading ? <>
                <div className="upload-spinner"/>
                <h3>Processingâ€¦</h3>
                <p>Extracting face embeddings from your files.</p>
              </> : <>
                <Upload size={48} strokeWidth={1.3}/>
                <h3>Upload photos or video</h3>
                <p>Drop image or video files here, or click to browse. Each file should contain one clearly visible face.</p>
                <button disabled={!consent || !studentId} onClick={e => { e.stopPropagation(); fileRef.current?.click(); }}>
                  <FileVideo size={16}/>Choose files
                </button>
              </>}
            </div>
            <p className="caption"><Info size={15}/> Accepted: JPEG, PNG images and MP4 video (max 10 MB each). Video is sampled at 1 frame/second.</p>
          </section>
        }
      </div>

      <form className="panel controls" onSubmit={save}>
        <h2>Student enrollment</h2>

        <label>Select student
          <select value={studentId} onChange={e => { setStudentId(e.target.value); setTokens([]); setResult(null); }}>
            <option value="">Choose a studentâ€¦</option>
            {(state?.students || []).map(s =>
              <option key={s.id} value={s.id}>{s.name} Â· {s.roll_no}{s.sample_count > 0 ? ' (already enrolled)' : ''}</option>
            )}
          </select>
        </label>

        {selectedStudent && <div className="selected-student-info">
          <span className="avatar">{selectedStudent.name?.[0]?.toUpperCase()}</span>
          <div>
            <b>{selectedStudent.name}</b>
            <small>{selectedStudent.roll_no} Â· {selectedStudent.section} Â· {selectedStudent.batch || 'â€”'}</small>
          </div>
        </div>}

        <label className="consent">
          <input type="checkbox" checked={consent} onChange={e => setConsent(e.target.checked)}/>
          <span>I have explained the local attendance demonstration and data deletion process. This student explicitly consents to face capture and embedding storage.</span>
        </label>

        <div className="divider"/>
        <div className="panel-heading"><h3>Guided capture</h3><span className="score">{tokens.length} / 5 minimum</span></div>
        <div className="progress-track"><div style={{width:`${Math.min(tokens.length / 5 * 100, 100)}%`}}/></div>
        <p className="capture-prompt">{tokens.length < 15 ? prompts[Math.min(tokens.length, 5)] : 'Capture complete. Save the enrollment.'}</p>

      </form>
    </div>
  </>;
}

import { useEffect, useRef, useState } from 'react';
import { Camera as CameraIcon, Video, Square, Info, SwitchCamera } from 'lucide-react';

export default function Camera({ onFrame, automatic = false, enabled = true, result, actionLabel = 'Capture sample' }) {
  const video = useRef(null), stream = useRef(null), canvas = useRef(null), handler = useRef(onFrame);
  const alive = useRef(true), running = useRef(false), busy = useRef(false);
  const [active, setActive] = useState(false), [working, setWorking] = useState(false), [error, setError] = useState('');

  // Default to back camera ('environment') on mobile devices for scanning, and 'user' on desktop
  const isMobile = typeof navigator !== 'undefined' && /Android|iPhone|iPad|iPod/i.test(navigator.userAgent || '');
  const [facingMode, setFacingMode] = useState(isMobile ? 'environment' : 'user');
  const [aspectRatio, setAspectRatio] = useState(isMobile ? '3 / 4' : '4 / 3');

  handler.current = onFrame;

  useEffect(() => {
    alive.current = true;
    return () => {
      alive.current = false;
      running.current = false;
      stream.current?.getTracks().forEach(t => t.stop());
    };
  }, []);

  const stop = () => {
    running.current = false;
    stream.current?.getTracks().forEach(t => t.stop());
    stream.current = null;
    setActive(false);
  };

  useEffect(() => {
    if (!enabled && active) stop();
  }, [enabled, active]);

  async function start(mode = facingMode) {
    setError('');
    setWorking(true);
    if (stream.current) {
      stream.current.getTracks().forEach(t => t.stop());
      stream.current = null;
    }
    try {
      const constraints = {
        video: {
          facingMode: { ideal: mode },
          width: { ideal: 1280 },
          height: { ideal: 720 },
        },
        audio: false,
      };
      let media;
      try {
        media = await navigator.mediaDevices.getUserMedia(constraints);
      } catch (err) {
        media = await navigator.mediaDevices.getUserMedia({ video: true, audio: false });
      }

      if (!alive.current) {
        media.getTracks().forEach(t => t.stop());
        return;
      }
      stream.current = media;
      if (video.current) {
        video.current.srcObject = media;
        await video.current.play();
      }
      running.current = true;
      setActive(true);
    } catch (e) {
      stream.current?.getTracks().forEach(t => t.stop());
      setError('Camera unavailable. Allow camera access and close other apps using it.');
    } finally {
      if (alive.current) setWorking(false);
    }
  }

  async function flipCamera() {
    const nextMode = facingMode === 'user' ? 'environment' : 'user';
    setFacingMode(nextMode);
    if (active) {
      await start(nextMode);
    }
  }

  const handleLoadedMetadata = () => {
    const v = video.current;
    if (v && v.videoWidth && v.videoHeight) {
      setAspectRatio(`${v.videoWidth} / ${v.videoHeight}`);
    }
  };

  async function capture() {
    if (busy.current || !running.current || !video.current?.videoWidth) return;
    busy.current = true;
    if (alive.current) setWorking(true);
    try {
      const v = video.current, c = canvas.current;
      const maxDim = Math.max(v.videoWidth, v.videoHeight);
      const scale = Math.min(1, 640 / maxDim);
      c.width = Math.round(v.videoWidth * scale);
      c.height = Math.round(v.videoHeight * scale);
      c.getContext('2d').drawImage(v, 0, 0, c.width, c.height);
      await handler.current(c.toDataURL('image/jpeg', 0.85));
      if (alive.current) setError('');
    } catch (e) {
      if (alive.current) setError(e.message);
    } finally {
      busy.current = false;
      if (alive.current) setWorking(false);
    }
  }

  useEffect(() => {
    if (!automatic || !active || !enabled) return;
    const timer = setInterval(capture, 450);
    return () => clearInterval(timer);
  }, [automatic, active, enabled]);

  return (
    <section className="panel camera-panel">
      <div className="panel-heading">
        <h2>Camera preview</h2>
        <div style={{ display: 'flex', alignItems: 'center', gap: '8px' }}>
          <span className={`status ${active ? 'online' : ''}`}>
            <i />
            {active ? `Live (${facingMode === 'environment' ? 'Rear' : 'Front'})` : 'Offline'}
          </span>
          <button
            type="button"
            className="icon-button"
            title="Switch front/rear camera"
            onClick={flipCamera}
            style={{ padding: '5px', cursor: 'pointer', borderRadius: '4px' }}
          >
            <SwitchCamera size={17} />
          </button>
        </div>
      </div>
      <div className="camera-stage" style={{ aspectRatio, maxHeight: '600px' }}>
        <video
          ref={video}
          muted
          playsInline
          onLoadedMetadata={handleLoadedMetadata}
          className={active ? '' : 'hidden'}
          style={{ width: '100%', height: '100%', objectFit: 'contain' }}
        />
        {!active && (
          <div className="camera-empty">
            <CameraIcon size={54} strokeWidth={1.3} />
            <h3>Your camera is off</h3>
            <p>{enabled ? 'Start the camera to begin scanning.' : 'Complete consent before starting the camera.'}</p>
            <button onClick={() => start(facingMode)} disabled={!enabled || working}>
              <Video size={17} />
              {working ? 'Opening cameraâ€¦' : 'Start camera'}
            </button>
          </div>
        )}
        <div className="frame-corners" aria-hidden="true"><i /><i /><i /><i /></div>
        {active && result?.box && (
          <div
            className={`face-box ${result.identity ? 'matched' : ''}`}
            style={{
              left: `${(result.box[0] / result.width) * 100}%`,
              top: `${(result.box[1] / result.height) * 100}%`,
              width: `${(result.box[2] / result.width) * 100}%`,
              height: `${(result.box[3] / result.height) * 100}%`,
            }}
          >
            <span>{result.name || (result.quality ? 'Face detected' : 'Adjust position')}</span>
          </div>
        )}
        {active && (
          <div className="camera-toolbar">
            <span>{working ? 'Scanningâ€¦' : result?.reason || 'Position face inside the frame'}</span>
            <div className="camera-toolbar-actions">
              <button type="button" className="camera-flip" onClick={flipCamera} title="Switch camera">
                <SwitchCamera size={13} /> Flip
              </button>
              <button type="button" className="camera-stop" onClick={stop}>
                <Square size={13} /> Stop
              </button>
            </div>
          </div>
        )}
      </div>
      <canvas ref={canvas} className="hidden" />
      {error && <div className="inline-error" role="alert">{error}</div>}
      {active && !automatic && (
        <button className="capture-button" disabled={!enabled || working} onClick={capture}>
          <CameraIcon size={17} />
          {working ? 'Checking qualityâ€¦' : actionLabel}
        </button>
      )}
      <p className="caption">
        <Info size={15} /> Frames are processed locally in real time.
      </p>
    </section>
  );
}

A Deep Learning capstone application built with React, FastAPI, YuNet and SFace. Real pretrained ONNX inference, landmark alignment, cosine matching, consent-based enrollment, persistent attendance and a validation/test accuracy lab. No fabricated students or accuracy numbers.

## Start on Windows

Prerequisites: Python 3.11+ and Node.js 20.19+ (only needed to build the UI).

```powershell
cd D:\capstone
.\setup.ps1
.\start.ps1
```

Open http://127.0.0.1:8000 in Chrome or Edge. If PowerShell execution policy prevents scripts, run `powershell -ExecutionPolicy Bypass -File .\start.ps1` for that invocation. Stop with Ctrl+C. After initial setup, `start.ps1` is enough. Models and frontend run locally without an internet connection. Bind to loopback only; do not expose this prototype to a network.

## First demonstration

1. Open **Volunteers â†’ Enroll volunteer**. Explain the purpose, storage and deletion procedure; obtain explicit consent.
2. Enter the participant name and student ID, record consent, then start the camera. Capture at least five distinct views, preferably 10â€“15 across modest pose and lighting variation. Save the participant. Raw images are not persisted. Unsaved embedding captures expire after ten minutes.
3. Enroll at least two volunteers. Only one person should appear in the camera at a time.
4. Open **Live attendance**, name and start a session, then start the camera. Three consecutive accepted observations trigger one record per volunteer/session. Returning or restarting does not create duplicates.
5. Test an additional consenting, unenrolled volunteer. The system should show Unknown rather than mark attendance. Incorrect acceptance is a failure to record and investigate, not a result to hide.
6. End the session. View **Attendance records** and export CSV. Timestamps are stored in UTC and displayed in browser local time.

## Accuracy comes from experiments

The default similarity threshold **0.45** and ambiguity margin **0.08** are provisional engineering choices, not measured guarantees. Enrollment creates normalized centroids from multiple normalized SFace embeddings. Matching takes the highest cosine similarity, rejects scores below threshold and rejects insufficient separation from the second-best identity. Three-frame consistency is not liveness detection.

**Accuracy lab** implements a local experiment:

- Collect labeled validation attempts from known and consenting unknown volunteers. Use a new capture session from enrollment. Choose ground truth before capture; incorrect labels invalidate results.
- Minimum five known and five unknown attempts permits calibration, but this is only a software minimum. Aim for substantially more attempts per condition and volunteer; report subject counts, lighting, pose and capture days.
- Calibrate once: select a threshold using maximum impostor scores per attempt, targeting observed validation false-match rate <=1%. A small dataset cannot substantiate a population-wide 1% rate. The ambiguity margin is fixed and may further reject attempts.
- Calibration freezes validation. Collect held-out test attempts in a different batch, using different unknown volunteers. Never adjust parameters after inspecting test outcomes without defining a new held-out test set.
- Reports include known identification accuracy (rejections count as failures), unknown rejection accuracy, wrong-identity count, pairwise verification accuracy/FAR/FRR with comparison counts, and median/p95 extraction latency. Pairwise comparisons are correlated; do not treat them as independent trials.
- Quality-failed captures are not saved as scored attempts. Log capture failures separately in your project report. Latency covers decode, detection and feature extraction; it excludes network round trip, gallery matching and multi-frame confirmation time.
- Adding/deleting volunteers resets calibration and probes because the gallery changed. Reset experiment deletes only probes/calibration, not volunteers or attendance.

Suggested comparisons: one versus several enrollment samples; frontal/even lighting versus pose/lighting variation; different thresholds; single-frame versus three-frame decisions. Keep separate held-out sets for final claims. The UI currently uses centroid matching and three-frame confirmation; alternative strategies should be added as explicit experiment configurations, not silently changed.

### LFW verification

LFW is not included or automatically downloaded. Obtain the original dataset and official `pairs.txt` from its source under applicable terms. Run:

```powershell
.\.venv\Scripts\python.exe scripts\evaluate_lfw.py --images D:\datasets\lfw --pairs D:\datasets\pairs.txt --output data\lfw_report.json
```

The script requires the official 6000-pair, ten-fold file. Each fold uses a threshold selected on the other nine folds. It reports quality/detection failures, end-to-end accuracy counting failed pairs as incorrect, and conditional verification metrics for valid pairs. It uses the local quality gate, so it is not an exact reproduction of published SFace scores. Pretraining overlap is not audited. An LFW result is not a substitute for local open-set testing.

## Architecture and files

```text
Browser camera (explicit start)
  â†’ JPEG over loopback â†’ image validation â†’ YuNet + facial landmarks
  â†’ face-size / blur / lighting checks â†’ aligned SFace embedding
  â†’ L2 normalization â†’ cosine centroid matching + ambiguity check
  â†’ consecutive-frame confirmation â†’ unique SQLite attendance entry
```

- `backend/vision.py`: actual models, image decoding, quality gates and alignment.
- `backend/core.py`: matching, temporal confirmation, threshold calibration and metrics.
- `backend/store.py`: SQLite persistence and atomic uniqueness constraints.
- `backend/main.py`: validated HTTP API, consent workflow, calibration, exports and static UI.
- `frontend/src/`: React interface and camera lifecycle.
- `scripts/download_models.py`: official OpenCV Zoo downloads with recorded SHA-256 provenance.
- `scripts/evaluate_lfw.py`: fold-separated benchmark runner.
- `tests/`: synthetic-vector logic tests, isolated API tests and real-model blank-frame smoke test.
- `data/`: local SQLite database; excluded from Git.

The backend is deliberately one process/worker because ephemeral capture tokens and observation history are in memory. SQLite has foreign keys, WAL, and a session/student unique key. Modules provide extension points for other models, matching strategies and a future database service. Scaling to multiple workers requires shared observation/token state and a security design; do not add `--workers` to this prototype.

## Privacy and limitations

- Embeddings are sensitive biometric data, not anonymous or guaranteed irreversible. Local SQLite is **not encrypted**; restrict OS access, avoid syncing/backing up biometric data without consent, and delete it at the agreed retention date.
- UI consent records an operator attestation with timestamp, not a legal consent document. Maintain an appropriate participant information/consent process separately.
- Deleting a participant deletes their embeddings and attendance rows. This is logical deletion, not a guarantee of forensic erasure from storage/backups.
- No raw frames are saved by the application. In-memory data is not claimed to be securely wiped.
- No liveness or presentation-attack protection. A photo or screen replay may be accepted. Use only consenting adults in supervised demonstrations; do not make real attendance or access-control decisions from this prototype.
- No login or multi-user administration. Same-origin and host restrictions reduce accidental exposure but are not authentication. Use only on a trusted local computer.
- Camera permissions and real volunteer end-to-end accuracy need to be verified on the target laptop. Automated synthetic tests do not establish biometric accuracy.

"""Model-independent decisions and evaluation. Similarities are not probabilities."""
import numpy as np


def normalize(vector):
    v = np.asarray(vector, dtype=np.float32).reshape(-1)
    norm = np.linalg.norm(v)
    if not np.isfinite(v).all() or norm < 1e-8:
        raise ValueError('Invalid embedding')
    return v / norm


def match(vector, gallery, threshold=.45, margin=.08):
    ranked = sorted(((key, float(np.clip(np.dot(vector, normalize(value)), -1, 1)))
                     for key, value in gallery.items()), key=lambda x: x[1], reverse=True)
    if not ranked:
        return dict(identity=None, score=None, gap=None, reason='No volunteers enrolled', scores={})
    key, score = ranked[0]
    gap = score - ranked[1][1] if len(ranked) > 1 else 2.
    accepted = score >= threshold and gap >= margin
    return dict(identity=key if accepted else None, score=score, gap=gap,
                reason='Match' if accepted else ('Ambiguous match' if score >= threshold else 'Unknown'),
                scores=dict(ranked))


class Stability:
    def __init__(self):
        self.identity = None
        self.count = 0
        self.last = 0.

    def observe(self, identity, now):
        self.count = self.count + 1 if identity and identity == self.identity and now-self.last < 2 else (1 if identity else 0)
        self.identity, self.last = identity, now
        return self.count >= 3


def verification_metrics(genuine, impostor, threshold):
    g, i = np.asarray(genuine), np.asarray(impostor)
    return dict(genuine_count=len(g), impostor_count=len(i),
                far=float(np.mean(i >= threshold)) if len(i) else None,
                frr=float(np.mean(g < threshold)) if len(g) else None,
                accuracy=float(((g >= threshold).sum() + (i < threshold).sum()) / (len(g)+len(i))) if len(g)+len(i) else None)


def calibrate(genuine, impostor, target_far=.01):
    if not len(genuine) or not len(impostor):
        raise ValueError('Both genuine and impostor validation attempts are required')
    # Include just-above endpoints so a zero observed FAR solution is possible.
    candidates = sorted(set([-1., 1.000001] + [float(x) for x in genuine+impostor] + [float(x)+1e-6 for x in impostor]))
    valid = [t for t in candidates if np.mean(np.asarray(impostor) >= t) <= target_far]
    return min(valid, key=lambda t: (np.mean(np.asarray(genuine) < t), t))

import {useEffect, useState} from 'react';
import {api} from '../api';

// Abort requests when a filter changes. Keyed state also prevents a one-frame stale report.
export function useResource(path, revision=0) {
  const key = `${path}:${revision}`;
  const [state,setState]=useState({key:null,data:null,error:null,loading:false});
  useEffect(()=>{
    if (!path) return;
    const controller=new AbortController();
    setState({key,data:null,error:null,loading:true});
    api(path,{signal:controller.signal}).then(data=>{
      if (!controller.signal.aborted) setState({key,data,error:null,loading:false});
    }).catch(error=>{
      if (!controller.signal.aborted) setState({key,data:null,error:error.message,loading:false});
    });
    return ()=>controller.abort();
  },[path,key]);
  return state.key===key ? state : {data:null,error:null,loading:!!path};
}

"""Run the official 10-fold LFW pairs file with frozen YuNet/SFace.

No dataset is downloaded. Supply lawfully obtained LFW images and pairs.txt.
Each fold's threshold is selected using the other nine folds only.
Detection/quality failures count as incorrect in end-to-end accuracy; conditional
verification metrics are additionally reported for successfully processed pairs.
"""
import argparse
import base64
import json
import sys
from pathlib import Path
import numpy as np

sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from backend.vision import FaceEngine
from backend.core import verification_metrics


def parse_pairs(path, root):
    lines=path.read_text().strip().splitlines()
    header=lines[0].split()
    if header != ['10','300']:
        raise ValueError('Expected official 10-fold pairs.txt header: 10 300')
    if len(lines)-1 != 6000: raise ValueError('Expected 6000 pairs')
    pairs=[]
    for index,line in enumerate(lines[1:]):
        fields=line.split()
        if len(fields)==3:
            name,a,b=fields
            left=root/name/f'{name}_{int(a):04d}.jpg'
            right=root/name/f'{name}_{int(b):04d}.jpg'
            same=True
        elif len(fields)==4:
            n1,a,n2,b=fields
            left=root/n1/f'{n1}_{int(a):04d}.jpg'
            right=root/n2/f'{n2}_{int(b):04d}.jpg'
            same=False
        else: raise ValueError(f'Invalid pair on line {index+2}')
        pairs.append((index//600,left,right,same))
    return pairs


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--images',type=Path,required=True)
    parser.add_argument('--pairs',type=Path,required=True)
    parser.add_argument('--output',type=Path,default=Path('data/lfw_report.json'))
    args=parser.parse_args()
    pairs=parse_pairs(args.pairs,args.images)
    missing={str(p) for _,a,b,_ in pairs for p in (a,b) if not p.exists()}
    if missing: raise ValueError(f'{len(missing)} images missing; first: {next(iter(missing))}')
    engine=FaceEngine()
    cache={}
    failures={}
    latency=[]
    def embed(path):
        if path not in cache:
            result=engine.extract(base64.b64encode(path.read_bytes()).decode())
            cache[path]=result.get('embedding')
            latency.append(result['latency_ms'])
            if cache[path] is None: failures[str(path)]=result['reason']
        return cache[path]
    observations=[]
    for i,(fold,a,b,same) in enumerate(pairs):
        va,vb=embed(a),embed(b)
        score=float(np.clip(np.dot(va,vb),-1,1)) if va is not None and vb is not None else None
        observations.append(dict(fold=fold,same=same,score=score))
        if (i+1)%100==0: print(f'{i+1}/6000 pairs',flush=True)
    folds=[]
    for fold in range(10):
        train=[r for r in observations if r['fold']!=fold and r['score'] is not None]
        test=[r for r in observations if r['fold']==fold]
        if not train: raise ValueError('No valid training-fold pairs')
        # Threshold sweep uses only nine development folds, not the held-out fold.
        thresholds=np.linspace(-1,1,2001)
        scores=np.array([r['score'] for r in train])
        labels=np.array([r['same'] for r in train])
        threshold=float(max(thresholds,key=lambda t:float(np.mean((scores>=t)==labels))))
        valid=[r for r in test if r['score'] is not None]
        correct=sum((r['score']>=threshold)==r['same'] for r in valid)
        folds.append(dict(fold=fold+1,threshold=threshold,pairs=len(test),failed_pairs=len(test)-len(valid),
                          end_to_end_accuracy=correct/len(test),
                          conditional_verification=verification_metrics([r['score'] for r in valid if r['same']],
                                                                        [r['score'] for r in valid if not r['same']],threshold)))
    report=dict(protocol='Official 10-fold pair definitions; frozen external pretrained models; quality gate enabled',
                model='YuNet 2023mar + SFace 2021dec',folds=folds,
                mean_end_to_end_accuracy=float(np.mean([f['end_to_end_accuracy'] for f in folds])),
                std_across_folds=float(np.std([f['end_to_end_accuracy'] for f in folds])),
                unique_images=len(cache),failed_images=len(failures),failure_reasons=failures,
                extraction_latency_median_ms=float(np.median(latency)),
                caveats=['Pretraining data overlap has not been independently audited.',
                         'Quality gates differ from published SFace benchmarks; do not compare as identical protocols.',
                         'Not a measure of local open-set attendance recognition.'])
    args.output.parent.mkdir(parents=True,exist_ok=True)
    args.output.write_text(json.dumps(report,indent=2))
    print(f'Report written to {args.output}')


if __name__=='__main__': main()

"""Framework-neutral logger for real epoch values and independent evaluation trials."""
import json
import sys
from pathlib import Path

sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from backend.lab import Run, Epoch, ModelInfo, Trial


class MetricsLogger:
    def __init__(self, class_id, student_id, name, model, training_images, validation_images):
        self.run=Run(class_id=class_id,student_id=student_id,name=name,model=ModelInfo(**model),
                     counts={'training_images':training_images,'validation_images':validation_images})

    def log_epoch(self,epoch,train_accuracy,validation_accuracy,train_loss,validation_loss):
        self.run.history.append(Epoch(epoch=epoch,train_accuracy=train_accuracy,
            validation_accuracy=validation_accuracy,train_loss=train_loss,validation_loss=validation_loss))

    def log_trial(self,actual_positive,predicted_positive,similarity):
        self.run.trials.append(Trial(actual_positive=actual_positive,
                                    predicted_positive=predicted_positive,similarity=similarity))

    def save(self,path,training_seconds=None,threshold=.45,notes=''):
        self.run.model.training_seconds=training_seconds
        self.run.threshold=threshold
        self.run.notes=notes
        # Revalidate the full payload before writing. No predictions or history are invented.
        validated=Run.model_validate(self.run.model_dump())
        target=Path(path)
        target.parent.mkdir(parents=True,exist_ok=True)
        temporary=target.with_suffix(target.suffix+'.tmp')
        temporary.write_text(json.dumps(validated.model_dump(),indent=2,allow_nan=False),encoding='utf-8')
        temporary.replace(target)
        return target

"""Password-protected LAN HTTPS gateway to the unchanged loopback application."""
import argparse
import base64
import hmac
import http.client
import ipaddress
import json
from pathlib import Path
import secrets

import subprocess

ROOT = Path(__file__).resolve().parents[1]
LOCAL = ROOT / '.local/mobile'


def make_handler(username, password, hosts, backend_port=8000):
    expected = 'Basic ' + base64.b64encode(f'{username}:{password}'.encode()).decode()
    class Gateway(BaseHTTPRequestHandler):
        def log_message(self, *args):
            pass

        def handle_request(self):
            host = self.headers.get('Host', '')
            if urlsplit('https://' + host).hostname not in hosts:
                self.send_error(403)
                return
            if not hmac.compare_digest(self.headers.get('Authorization', '').encode(), expected.encode()):
                self.send_response(401)
                self.send_header('WWW-Authenticate', 'Basic realm="Attend local prototype"')
                self.send_header('Cache-Control', 'no-store')
                self.end_headers()
                return
            origin = self.headers.get('Origin')
            if origin and origin != 'https://' + host:
                self.send_error(403)
                return
            if self.headers.get('Transfer-Encoding') or not self.path.startswith('/') or self.path.startswith('//'):
                self.send_error(400)
                return
            try:
                length = int(self.headers.get('Content-Length', '0'))
            except ValueError:
                self.send_error(400)
                return
            if not 0 <= length <= 10_000_000:
                self.send_error(413)
                return
            self.connection.settimeout(60)
            data = self.rfile.read(length) if length else None
            headers = {k:v for k,v in self.headers.items() if k.lower() not in {'host','origin','authorization','connection','transfer-encoding','forwarded','x-forwarded-host','x-forwarded-proto'}}
            headers['Host'] = f'127.0.0.1:{backend_port}'
            if origin:
                headers['Origin'] = f'http://127.0.0.1:{backend_port}'
            conn = http.client.HTTPConnection('127.0.0.1', backend_port, timeout=60)
            try:
                conn.request(self.command, self.path, body=data, headers=headers)
                response = conn.getresponse()
                body = response.read()
                self.send_response(response.status)
                for key,value in response.getheaders():
                    if key.lower() not in {'connection','transfer-encoding','server','date','content-length'}:
                        self.send_header(key,value)
                self.send_header('Content-Length', str(len(body)))
                self.end_headers()
                if self.command != 'HEAD':
                    self.wfile.write(body)
            except (OSError, http.client.HTTPException):
                self.send_error(502, 'Start the local application using start.ps1')
            finally:
                conn.close()
        do_GET = do_POST = do_PUT = do_DELETE = do_PATCH = do_HEAD = handle_request
    return Gateway


def initialize(address):
    from datetime import datetime, timedelta, timezone
    from cryptography import x509
    from cryptography.hazmat.primitives import hashes, serialization
    from cryptography.hazmat.primitives.asymmetric import rsa
    from cryptography.x509.oid import NameOID, ExtendedKeyUsageOID
    ipaddress.ip_address(address)
    LOCAL.mkdir(parents=True, exist_ok=True)
    config_path = LOCAL/'config.json'
    config = json.loads(config_path.read_text()) if config_path.exists() else {'username':'student','password':secrets.token_urlsafe(24)}
    config['address'] = address
    key_path = LOCAL/'ca.key'
    now = datetime.now(timezone.utc)
    if key_path.exists():
        key = serialization.load_pem_private_key(key_path.read_bytes(), password=None)
        ca = x509.load_pem_x509_certificate((LOCAL/'ca.pem').read_bytes())
    else:
        key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
        name = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, 'Attend Laptop Local CA')])
        ca = (x509.CertificateBuilder().subject_name(name).issuer_name(name).public_key(key.public_key())
              .serial_number(x509.random_serial_number()).not_valid_before(now-timedelta(minutes=5))
              .not_valid_after(now+timedelta(days=365)).add_extension(x509.BasicConstraints(ca=True,path_length=0),True)
              .add_extension(x509.KeyUsage(False,False,False,False,False,True,True,False,False),True).sign(key,hashes.SHA256()))
        key_path.write_bytes(key.private_bytes(serialization.Encoding.PEM,serialization.PrivateFormat.PKCS8,serialization.NoEncryption()))
        (LOCAL/'ca.pem').write_bytes(ca.public_bytes(serialization.Encoding.PEM))
    (LOCAL/'attend-root.cer').write_bytes(ca.public_bytes(serialization.Encoding.DER))
    server_key = rsa.generate_private_key(public_exponent=65537,key_size=2048)
    cert = (x509.CertificateBuilder().subject_name(x509.Name([x509.NameAttribute(NameOID.COMMON_NAME,address)]))
            .issuer_name(ca.subject).public_key(server_key.public_key()).serial_number(x509.random_serial_number())
            .not_valid_before(now-timedelta(minutes=5)).not_valid_after(now+timedelta(days=90))
            .add_extension(x509.SubjectAlternativeName([x509.IPAddress(ipaddress.ip_address(address)),x509.IPAddress(ipaddress.ip_address('127.0.0.1')),x509.DNSName('localhost')]),False)
            .add_extension(x509.BasicConstraints(ca=False,path_length=None),True)
            .add_extension(x509.ExtendedKeyUsage([ExtendedKeyUsageOID.SERVER_AUTH]),False).sign(key,hashes.SHA256()))
    (LOCAL/'server.pem').write_bytes(cert.public_bytes(serialization.Encoding.PEM))
    (LOCAL/'server.key').write_bytes(server_key.private_bytes(serialization.Encoding.PEM,serialization.PrivateFormat.PKCS8,serialization.NoEncryption()))
    config_path.write_text(json.dumps(config,indent=2))
    (LOCAL/'PHONE-SETUP.txt').write_text(f'Open https://{address}:8443 after installing attend-root.cer as a trusted local CA on your phone.\nUsername: {config["username"]}\nPassword: {config["password"]}\nKeep this file and all .key files private. Share only attend-root.cer with your own test phones.\n')
    print(f'HTTPS configured for https://{address}:8443. Credentials are in .local/mobile/PHONE-SETUP.txt')


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--init', metavar='LAN_IP')
    args = parser.parse_args()
    if args.init:
        initialize(args.init)
        return
    config = json.loads((LOCAL/'config.json').read_text())
    server = ThreadingHTTPServer((config['address'],8443),make_handler(config['username'],config['password'],[config['address'],'localhost','127.0.0.1']))
    context = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
    context.minimum_version = ssl.TLSVersion.TLSv1_2
    context.load_cert_chain(LOCAL/'server.pem', LOCAL/'server.key')
    server.socket = context.wrap_socket(server.socket,server_side=True)
    print(f'Attend mobile gateway: https://{config["address"]}:8443',flush=True)
    try:
        server.serve_forever()
    finally:
        server.server_close()


if __name__ == '__main__':
    main()

    # Back up old database if it exists
    old = Path(db_path)
    if old.exists():
        backup = old.with_suffix('.db.bak')
        if not backup.exists():
            import shutil
            shutil.copy2(old, backup)
            print(f'Backed up existing database to {backup}')
        old.unlink()
        # Also remove WAL/SHM files
        for suffix in ('.db-wal', '.db-shm'):
            p = Path(str(old) + suffix.replace('.db', ''))
            if p.exists():
                p.unlink()
        print('Removed old database for fresh schema')

    store = Store(db_path)

    # Import students
    roster_path = ROOT / 'data' / 'imports' / 'section-d-students.json'
    if not roster_path.exists():
        print(f'Student roster not found at {roster_path}')
        return

    students = json.loads(roster_path.read_text(encoding='utf-8'))
    count = 0
    for s in students:
        try:
            store.add_student(
                roll_no=s['roll_no'],
                name=s['name'],
                email=s.get('email'),
                section=s.get('section', 'D'),
                batch=s.get('batch'),
                photo_path=s.get('photo_path')
            )
            count += 1
        except Exception as e:
            print(f'  Skipped {s["roll_no"]}: {e}')
    print(f'Imported {count} students')

    # Import timetable
    tt_count = 0
    for rule in TIMETABLE:
        store.add_timetable_rule(**rule)
        tt_count += 1
    print(f'Imported {tt_count} timetable rules')

    print('Seed complete.')

import base64
import http.client
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from scripts.mobile_server import make_handler


class Upstream(BaseHTTPRequestHandler):
    def do_GET(self):
        self.send_response(200)
        self.end_headers()
        self.wfile.write(self.headers.get('Host', '').encode())
    def log_message(self, *args):
        pass


def test_mobile_auth_origin_and_proxy():
    backend = ThreadingHTTPServer(('127.0.0.1', 0), Upstream)
    gateway = ThreadingHTTPServer(('127.0.0.1', 0), make_handler('student', 'test-password', ['localhost'], backend.server_port))
    for server in (backend, gateway):
        threading.Thread(target=server.serve_forever, daemon=True).start()
    def request(headers):
        conn = http.client.HTTPConnection('127.0.0.1', gateway.server_port)
        conn.request('GET', '/api/state', headers={'Host': f'localhost:{gateway.server_port}', **headers})
        response = conn.getresponse()
        result = response.status, response.read()
        conn.close()
        return result
    token = 'Basic ' + base64.b64encode(b'student:test-password').decode()
    try:
        assert request({})[0] == 401
        assert request({'Authorization': 'Basic bad'})[0] == 401
        assert request({'Authorization': token, 'Origin': 'https://evil.example'})[0] == 403
        status, body = request({'Authorization': token, 'Origin': f'https://localhost:{gateway.server_port}'})
        assert status == 200
        assert body == f'127.0.0.1:{backend.server_port}'.encode()
        assert request({'Authorization': token, 'Host': 'evil.example'})[0] == 403
    finally:
        for server in (gateway, backend):
            server.shutdown()
            server.server_close()
